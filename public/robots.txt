User-agent: *
Allow: /

# The admin panel is reachable by URL but linked from nowhere in the
# storefront. No reason for it in an index.
Disallow: /*/admin
Disallow: /*/admin/
Disallow: /admin

# Order pages carry a reference and a phone number in the query. Nothing to
# index, and the reference would otherwise leak into a search result.
Disallow: /*/order
Disallow: /*/order/
Disallow: /api/

# The prerendered copies behind the pretty URLs. They are served by a rewrite,
# not linked, and .htaccess answers a direct request with 404 — this is only
# here so a crawler that somehow finds the path does not index a duplicate.
Disallow: /_pages/

# Faceted and stateful views of the catalogue. Each is the same products as
# /products, which is the URL that should rank for them.
Disallow: /*/search
Disallow: /*/search?
Disallow: /*/cart
Disallow: /*/checkout

# The storefront is a static SPA with no server-side routes, so the sitemap is
# generated by the API and proxied through this host by .htaccess. It is listed
# here under chammastore.com rather than under the API because that is the only
# host Search Console will accept it on.
Sitemap: https://chammastore.com/sitemap.xml
